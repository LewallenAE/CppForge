cmake_minimum_required(VERSION 3.20)
project(CinderDB VERSION 1.0.0 LANGUAGES CXX)

option(CINDERDB_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(CINDERDB_ENABLE_TSAN "Enable ThreadSanitizer" OFF)
if(CINDERDB_ENABLE_SANITIZERS AND CINDERDB_ENABLE_TSAN)
    message(FATAL_ERROR "Use separate ASan/UBSan and TSan build directories")
endif()

find_package(Threads REQUIRED)
add_library(cinderdb_options INTERFACE)
target_compile_features(cinderdb_options INTERFACE cxx_std_20)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(cinderdb_options INTERFACE -Wall -Wextra -Wconversion -Wsign-conversion -Werror -pedantic-errors)
    if(CINDERDB_ENABLE_SANITIZERS)
        target_compile_options(cinderdb_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(cinderdb_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endif()
    if(CINDERDB_ENABLE_TSAN)
        target_compile_options(cinderdb_options INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
        target_link_options(cinderdb_options INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
    endif()
elseif(CINDERDB_ENABLE_SANITIZERS OR CINDERDB_ENABLE_TSAN)
    message(FATAL_ERROR "Sanitizers require GCC or Clang")
endif()

add_library(cinderdb_core src/protocol.cpp src/wal.cpp src/store.cpp src/server.cpp)
target_include_directories(cinderdb_core PUBLIC include)
target_link_libraries(cinderdb_core PUBLIC cinderdb_options Threads::Threads)
set_target_properties(cinderdb_core PROPERTIES CXX_EXTENSIONS OFF)

add_executable(cinderdb src/main.cpp)
target_link_libraries(cinderdb PRIVATE cinderdb_core)
set_target_properties(cinderdb PROPERTIES CXX_EXTENSIONS OFF)

include(CTest)
if(BUILD_TESTING)
 add_executable(cinderdb_tests tests/cinderdb_tests.cpp)
 target_link_libraries(cinderdb_tests PRIVATE cinderdb_core)
 set_target_properties(cinderdb_tests PROPERTIES CXX_EXTENSIONS OFF)
 add_test(NAME cinderdb_all COMMAND cinderdb_tests)
 set_tests_properties(cinderdb_all PROPERTIES TIMEOUT 90)
endif()
